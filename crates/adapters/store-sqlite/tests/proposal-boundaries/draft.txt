use application::ProposalRepository;
use domain::{Draft, DraftRestoreEvidence, MasterInstrument, Proposal, ProposalId};
pub async fn use_draft<R: ProposalRepository>(repo: &R, value: &Proposal<Draft>, id: &ProposalId, evidence: &DraftRestoreEvidence, master: &MasterInstrument) {
    let _saved = repo.store_draft(value, evidence, master).await;
    let _loaded: Option<Proposal<Draft>> = repo.draft(id, master).await.expect("load");
}
fn main() {}
